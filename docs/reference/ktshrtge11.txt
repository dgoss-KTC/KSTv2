function evalemail returns logical (input addr as char):
 def var atsym as int.
 def var period as int.

 if addr = "" then do:
     return false.
 end.

 atsym = index(addr, "@", 2).
 period = index(addr, ".", atsym + 2).

 if atsym > 1 and period > 3 then
     return true.
 else
     return false.

end function.

procedure showHours:

define variable open_ref like wo_qty_ord label "Qty Open".

                      and wo_nbr = mstr_wkfl.mstr_nbr

 end.

   "W/O Nbr: " + string(mstr_wkfl.mstr_nbr) +
